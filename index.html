/* ฟังก์ชันบันทึกภาพ ปรับแก้ตัวหนังสือเลื่อนไม่ตรงบรรทัด */ function saveAsJPG(btnElement) { const srcCard = document.getElementById('quotation-card'); const originalBtnText = btnElement.innerHTML; btnElement.innerHTML = "⏳ กำลังสร้างรูปภาพ..."; btnElement.disabled = true; // 1. คัดลอก Card ออกมา const clone = srcCard.cloneNode(true); // 2. ซ่อนองค์ประกอบที่ไม่ต้องการบันทึกลงภาพ const ignoreElements = clone.querySelectorAll('[data-html2canvas-ignore="true"], .btn-del, .btn-add, .no-print'); ignoreElements.forEach(el => el.remove()); // 3. แปลง input/select/textarea เดิมให้เป็น span ข้อความธรรมดาที่ล็อกจัดตำแหน่งบรรทัด const srcInputs = srcCard.querySelectorAll('input, select, textarea'); const cloneInputs = clone.querySelectorAll('input, select, textarea'); srcInputs.forEach((srcInput, idx) => { const cloneInput = cloneInputs[idx]; if (!cloneInput) return; let valText = srcInput.value; if (srcInput.tagName === 'SELECT') { valText = srcInput.options[srcInput.selectedIndex] ? srcInput.options[srcInput.selectedIndex].text : srcInput.value; } if (srcInput.type === 'checkbox') { if (srcInput.checked) { const checkSpan = document.createElement('span'); checkSpan.innerText = '✓ '; checkSpan.style.fontWeight = 'bold'; checkSpan.style.color = '#2563eb'; checkSpan.style.display = 'inline-block'; checkSpan.style.verticalAlign = 'middle'; cloneInput.parentNode.insertBefore(checkSpan, cloneInput); } cloneInput.remove(); return; } const fontSpan = document.createElement('span'); fontSpan.innerText = valText; const comp = window.getComputedStyle(srcInput); // ล็อกคุณสมบัติเพื่อให้ข้อความจัดตรงกลางบรรทัดเดียวกับข้อความข้างเคียง fontSpan.style.fontFamily = "'Prompt', sans-serif"; fontSpan.style.fontSize = comp.fontSize; fontSpan.style.fontWeight = "700"; fontSpan.style.color = comp.color !== 'rgba(0, 0, 0, 0)' && comp.color !== 'transparent' ? comp.color : '#0f172a'; fontSpan.style.display = 'inline-flex'; fontSpan.style.alignItems = 'center'; fontSpan.style.justifyContent = srcInput.classList.contains('text-right') ? 'flex-end' : 'flex-start'; fontSpan.style.width = comp.width; fontSpan.style.height = comp.height; fontSpan.style.padding = comp.padding; fontSpan.style.margin = '0'; fontSpan.style.lineHeight = '1'; fontSpan.style.verticalAlign = 'middle'; fontSpan.style.boxSizing = 'border-box'; fontSpan.style.whiteSpace = 'pre-wrap'; cloneInput.parentNode.replaceChild(fontSpan, cloneInput); }); // 4. นำ Clone ไปวางซ่อนไว้นอกหน้าจอ const wrapper = document.createElement('div'); wrapper.style.position = 'absolute'; wrapper.style.left = '-9999px'; wrapper.style.top = '0'; wrapper.style.width = srcCard.offsetWidth + 'px'; wrapper.appendChild(clone); document.body.appendChild(wrapper); // 5. เรนเดอร์เป็นภาพ setTimeout(() => { html2canvas(clone, { scale: 2.5, useCORS: true, backgroundColor: "#ffffff", logging: false }).then(canvas => { const imageJPG = canvas.toDataURL("image/jpeg", 0.95); const link = document.createElement('a'); link.download = `ใบเสนอราคา_${new Date().toISOString().slice(0,10)}.jpg`; link.href = imageJPG; link.click(); document.body.removeChild(wrapper); btnElement.innerHTML = originalBtnText; btnElement.disabled = false; }).catch(err => { console.error("Save image failed: ", err); if (document.body.contains(wrapper)) document.body.removeChild(wrapper); alert("เกิดข้อผิดพลาดในการบันทึกภาพ"); btnElement.innerHTML = originalBtnText; btnElement.disabled = false; }); }, 150); }
